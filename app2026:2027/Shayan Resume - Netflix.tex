\newcommand{\resumeSubheading}[4]{
  \vspace{-2pt}\item
    \begin{tabular*}{0.97\textwidth}[t]{l@{\extracolsep{\fill}}r}
      \textbf{#1} & \textbf{#2} \\   % date BOLD: same font run as the title, so the
                                     % column jump is not split by a \Tf switch
      \textit{\small#3} & \textit{\small #4} \\
    \end{tabular*}\vspace{-7pt}
}

% One-line entry: title --- company on the left, date on the right. Use for roles
% with no bullets, where the standard two-line \resumeSubheading would leave a
% visible blank and its trailing \vspace{-7pt} would collide with the next heading.
\newcommand{\resumeSubheadingOneLine}[3]{
  \vspace{-2pt}\item
    \begin{tabular*}{0.97\textwidth}[t]{l@{\extracolsep{\fill}}r}
      \textbf{#1 --- #2} & \textbf{#3} \\   % all bold: no font switch anywhere on the line
    \end{tabular*}\vspace{-3pt}
}

\newcommand{\resumeProjectHeading}[2]{
    \item
    \begin{tabular*}{0.97\textwidth}{l@{\extracolsep{\fill}}r}
      \small#1 & \small\emph{#2} \\   % date must match #1's font AND SIZE. \small applies
                                     % only inside its own cell, so an unsized #2 landed at
                                     % normalsize -- a different font -> \Tf switch -> lost space.
    \end{tabular*}\vspace{-7pt}
}
